```ts
export class WidgetRegistry {
  private _widgets: { [type: string]: Widget<FormProperty, DFUISchemaItem> } =
    {};

  private defaultWidget!: Widget<FormProperty, DFUISchemaItem>;

  get widgets(): { [type: string]: Widget<FormProperty, DFUISchemaItem> } {
    return this._widgets;
  }

  setDefault(widget: NzSafeAny): void {
    this.defaultWidget = widget;
  }

  register(type: string, widget: NzSafeAny): void {
    this._widgets[type] = widget;
  }

  has(type: string): boolean {
    return this._widgets.hasOwnProperty(type);
  }

  getType(type: string): Widget<FormProperty, DFUISchemaItem> {
    if (this.has(type)) {
      return this._widgets[type];
    }
    return this.defaultWidget;
  }
}

@Injectable()
export class WidgetFactory {
  constructor(
    private registry: WidgetRegistry,
    private resolver: ComponentFactoryResolver
  ) {}

  createWidget(
    container: ViewContainerRef,
    type: string
  ): ComponentRef<Widget<FormProperty, DFUISchemaItem>> {
    if (!this.registry.has(type)) {
      console.warn(`No widget for type "${type}"`);
    }
    const componentClass = this.registry.getType(type) as NzSafeAny;
    const componentFactory =
      this.resolver.resolveComponentFactory<
        Widget<FormProperty, DFUISchemaItem>
      >(componentClass);
    return container.createComponent(componentFactory);
  }
}
```

```ts
export function useFactory(
  schemaValidatorFactory: SchemaValidatorFactory,
  cogSrv: DsmConfigService
): FormPropertyFactory {
  return new FormPropertyFactory(schemaValidatorFactory, cogSrv);
}

@Component({
  selector: "tl-df",
  templateUrl: "./df.component.html",
  providers: [
    WidgetFactory,
    {
      provide: FormPropertyFactory,
      useFactory,
      deps: [SchemaValidatorFactory, DsmConfigService],
    },
    TerminatorService,
  ],
  host: {
    "[class.tl-df__container]": "true",
    "[class.df]": "true",
    "[class.df__inline]": `layout === 'inline'`,
    "[class.df__horizontal]": `layout === 'horizontal'`,
    "[class.df__search]": `mode === 'search'`,
    "[class.df__edit]": `mode === 'edit'`,
    "[class.df__no-error]": `onlyVisual`,
    "[class.df__no-colon]": `noColon`,
    "[class.df__compact]": `compact`,
  },
})
export class TlDfComponent implements OnInit, OnChanges, OnDestroy {
  private destroy$ = new Subject<void>();
  private _item: Record<string, unknown>;
  private _renders = new Map<string, TemplateRef<void>>();
  private _valid = true;
  private _defUi!: DFUISchemaItem;
  readonly options: DsmDFConfig;
  private unsubscribe$ = new Subject<void>();

  _inited = false;
  locale: LocaleData = {};
  rootProperty: FormProperty | null = null;
  _formData: Record<string, unknown>;
  _btn: DFButton;
  _schema: DFSchema;
  _ui: DFUISchema;

  @Output() readonly formValueChange = new EventEmitter<DFValueChange>();
  @Output() readonly formChange = new EventEmitter<Record<string, unknown>>();
  @Output() readonly formSubmit = new EventEmitter<Record<string, unknown>>();
  @Output() readonly formReset = new EventEmitter<Record<string, unknown>>();
  @Output() readonly formError = new EventEmitter<ErrorData[]>();
  @Output() readonly formItemVisiableChange = new EventEmitter<{
    path: string;
    visible: boolean;
  }>();

  onSubmit(e: Event): void {
    e.preventDefault();
    e.stopPropagation();
    this.formSubmit.emit(this.value);
  }

  /**
   *
   * 表单是否有效
   */
  get valid(): boolean {
    return this._valid;
  }

  /**
   *
   * 表单值
   */
  get value(): { [key: string]: NzSafeAny } {
    return this._item;
  }

  /**
   *
   * 根据[路径]获取表单元素属性
   */
  getProperty(path: string): FormProperty | null {
    return this.rootProperty!.searchProperty(path);
  }

  /**
   *
   * 根据[路径]获取表单元素值
   */
  getValue(path: string): NzSafeAny {
    return this.getProperty(path)!.value;
  }

  /**
   *
   * 根据[路径]设置某个表单元素属性值
   */
  setValue(path: string, value: NzSafeAny): this {
    const item = this.getProperty(path);
    if (!item) {
      throw new Error(`Invalid path: ${path}`);
    }
    item.resetValue(value, { emitValidator: true, onlySelf: false });
    return this;
  }

  constructor(
    private formPropertyFactory: FormPropertyFactory,
    private terminator: TerminatorService,
    private dom: DomSanitizer,
    private cdr: ChangeDetectorRef,
    private localeSrv: DsmLocaleService,
    cogSrv: DsmConfigService,
    private platform: Platform
  ) {
    this.options = mergeConfig(cogSrv);
    this.autocomplete = this.options.autocomplete as "on" | "off";

    this.localeSrv.change.pipe(takeUntil(this.unsubscribe$)).subscribe(() => {
      this.locale = this.localeSrv.getData("df");
      if (this._inited) {
        this.cdr.markForCheck();
      }
    });
  }

  /**
   * 刷新schema
   */
  refreshSchema(newSchema?: DFSchema, newUI?: DFUISchema): this {
    if (!this.platform.isBrowser) {
      return this;
    }
    if (newSchema) {
      this.schema = newSchema;
    }
    if (newUI) {
      this.ui = newUI;
    }
    this.schema.type = "object";
    this._formData = { ...this.formData };
    if (this._inited) this.terminator.destroy();

    this.cleanRootSub();
    this.coverProperty();
    this.coverButtonProperty();
    // 创建rootProperty
    this.rootProperty = this.formPropertyFactory.createProperty(
      this._schema,
      this._ui,
      this.formData
    );
    this.attachCustomRender();
    this.cdr.detectChanges();
    this.reset();

    let isFirst = true;
    this.rootProperty.valueChanges.subscribe((res) => {
      this._item = {
        ...(this.cleanValue ? null : this.formData),
        ...res.value,
      };
      if (isFirst) {
        isFirst = false;
        return;
      }
      this.formChange.emit(this._item);
      this.formValueChange.emit({
        value: this._item,
        path: res.path,
        pathValue: res.pathValue,
      });
    });
    this.rootProperty.visiableChanges.subscribe((res) => {
      res.path && this.formItemVisiableChange.emit(res);
    });

    this.rootProperty.errorsChanges.subscribe((errors) => {
      this._valid = !(errors && errors.length);
      this.formError.emit(errors!);
      this.cdr.detectChanges();
    });

    return this;
  }

  /**
   *
   * 重置表单
   * @param [emitValidator] 是否触发校验，默认：`true`
   * @param [emit] 是否触发 `formReset` 事件，默认：`false`
   */
  reset(emitValidator = true, emit: boolean = false): this {
    if (!this.platform.isBrowser) {
      return this;
    }
    this.rootProperty!.resetValue(this.formData, {
      emitValidator,
      onlySelf: false,
    });
    Promise.resolve().then(() => this.cdr.detectChanges());
    if (emit) {
      this.formReset.emit(this.value);
    }
    return this;
  }

  /**
   * 手动校验所有表单项，常用于提交前的最后校验。并返回校验结果
   *
   */
  validatorAllItems(): boolean {
    this.rootProperty.validatorItem();
    return this._valid;
  }

  private inheritUI(ui: DFUISchemaItemRun): void {
    ["optionalHelp"]
      .filter((key) => !!this._defUi[key])
      .forEach((key) => (ui[key] = { ...this._defUi[key], ...ui[key] }));
    ["optionalHelpRight"]
      .filter((key) => !!this._defUi[key])
      .forEach((key) => (ui[key] = { ...this._defUi[key], ...ui[key] }));
  }

  /**
   * schema 属性补全
   */
  private coverProperty(): void {}

  private coverButtonProperty(): void {}

  /** 自定义模板渲染 */
  private attachCustomRender(): void {
    this._renders.forEach((tpl, path) => {
      const property = this.rootProperty?.searchProperty(path);
      if (property == null) {
        return;
      }
      property.ui._render = tpl;
    });
  }

  ngOnInit(): void {
    if (!this.platform.isBrowser) {
      return;
    }
    this.validator();
    this._inited = true;
  }

  ngOnChanges(
    changes: { [P in keyof this]?: SimpleChange } & SimpleChanges
  ): void {
    if (!this.platform.isBrowser) {
      return;
    }
    if (
      Object.keys(changes).length === 1 &&
      (changes.loading || changes.disabled)
    ) {
      this.cdr.detectChanges();
      return;
    }
    this.refreshSchema();
  }

  /**
   * 取消表单根节点的订阅事件
   * @returns
   */
  private cleanRootSub(): void {
    if (!this.rootProperty) return;
    this.rootProperty.errorsChanges.unsubscribe();
    this.rootProperty.valueChanges.unsubscribe();
  }

  ngOnDestroy(): void {
    this.terminator.destroy();
    const { unsubscribe$ } = this;
    unsubscribe$.next();
    unsubscribe$.complete();
  }
}
```

```ts
@Component({
  selector: "df-item",
  exportAs: "dfItem",
  host: { "[class.df__item]": "true" },

    <ng-template #target></ng-template>
    <ng-container *ngTemplateOutlet="footer"></ng-container>

})
export class DFItemComponent implements OnInit, OnChanges, OnDestroy {
  private ref!: ComponentRef<Widget<FormProperty, DFUISchemaItem>>;
  readonly unsubscribe$ = new Subject<void>();
  widget: Widget<FormProperty, DFUISchemaItem> | null = null;

  @Input() formProperty!: FormProperty;
  @Input() footer: TemplateRef<void> | null = null;

  @ViewChild("target", { read: ViewContainerRef, static: true })
  private container!: ViewContainerRef;

  constructor(
    private widgetFactory: WidgetFactory,
    private terminator: TerminatorService
  ) {}

  onWidgetInstanciated(widget: Widget<FormProperty, DFUISchemaItem>): void {
    this.widget = widget;
    const id = `_df-${nextUniqueId++}`;

    const ui = this.formProperty.ui as DFUISchemaItem;
    this.widget.formProperty = this.formProperty;
    this.widget.schema = this.formProperty.schema;
    this.widget.ui = ui;
    this.widget.id = id;
    this.formProperty.widget = widget;
  }

  ngOnInit(): void {
    this.terminator.onDestroy.subscribe(() => this.ngOnDestroy());
  }

  ngOnChanges(): void {
    const p = this.formProperty;

    // 这里是动态创建组件的核心代码，参考上面的widget.factory.ts
    this.ref = this.widgetFactory.createWidget(
      this.container,
      (p.ui.widget || p.schema.type) as string
    );
    this.onWidgetInstanciated(this.ref.instance);
  }

  ngOnDestroy(): void {
    const { unsubscribe$ } = this;
    unsubscribe$.next();
    unsubscribe$.complete();
    this.ref.destroy();
  }
}
```

```ts
import {
  AfterViewInit,
  Component,
  ElementRef,
  OnInit,
  ViewChild,
  ViewEncapsulation,
} from "@angular/core";
import { Subject } from "rxjs";
import {
  catchError,
  debounceTime,
  distinctUntilChanged,
  switchMap,
  takeUntil,
} from "rxjs/operators";

import type { NzSafeAny } from "ng-zorro-antd/core/types";
import { DFSchemaEnum } from "../../../types";
import { DFValue } from "../../../interface";
import { ControlUIWidget } from "../../../widget";
import { DFNzSelectWidgetSchema } from "./schema";
import { getData, toBool } from "../../../utils";
import { ArrayService } from "topdsm-lib/core/service";

@Component({
  selector: "df-nzSelect",
  templateUrl: "./select.widget.html",
  preserveWhitespaces: false,
  encapsulation: ViewEncapsulation.None,
})
export class SelectWidget
  extends ControlUIWidget<DFNzSelectWidgetSchema>
  implements OnInit, AfterViewInit
{
  private search$ = new Subject<string>();
  i: DFNzSelectWidgetSchema;
  data: DFSchemaEnum[];
  _value: NzSafeAny;
  hasGroup = false;
  loading = false;

  @ViewChild("wrap")
  wrap: ElementRef;

  private checkGroup(list: DFSchemaEnum[]): void {
    this.hasGroup = (list || []).filter((w) => w.group === true).length > 0;
  }

  ngOnInit(): void {
    const {
      autoClearSearchValue,
      borderless,
      autoFocus,
      dropdownMatchSelectWidth,
      serverSearch,
      maxMultipleCount,
      mode,
      showSearch,
      tokenSeparators,
      maxTagCount,
      compareWith,
      optionHeightPx,
      optionOverflowSize,
      showArrow,
    } = this.ui;
    this.i = {
      autoClearSearchValue: toBool(autoClearSearchValue, true),
      borderless: toBool(borderless, false),
      autoFocus: toBool(autoFocus, false),
      dropdownMatchSelectWidth: toBool(dropdownMatchSelectWidth, true),
      serverSearch: toBool(serverSearch, false),
      maxMultipleCount: maxMultipleCount || Infinity,
      mode: mode || "default",
      showSearch: toBool(showSearch, true),
      tokenSeparators: tokenSeparators || [],
      maxTagCount: maxTagCount || undefined,
      optionHeightPx: optionHeightPx || 30,
      optionOverflowSize: optionOverflowSize || 8,
      showArrow: typeof showArrow !== "boolean" ? undefined : showArrow,
      compareWith: compareWith || ((o1: NzSafeAny, o2: NzSafeAny) => o1 === o2),
    };

    const onSearch = this.ui.onSearch!;
    if (onSearch) {
      this.search$
        .pipe(
          takeUntil(this.sfItemComp!.unsubscribe$),
          distinctUntilChanged(),
          debounceTime(this.ui.searchDebounceTime || 300),
          switchMap((text) => onSearch(text)),
          catchError(() => [])
        )
        .subscribe((list) => {
          this.data = list;
          this.checkGroup(list);
          this.loading = false;
          this.detectChanges();
        });
    }
  }

  ngAfterViewInit(): void {}

  reset(value: DFValue): void {
    //参考 util.ts
    getData(this.schema, this.ui, value).subscribe((list) => {
      this._value = value;
      this.data = list;
      this.checkGroup(list);
      this.detectChanges();
    });
  }

  change(values: DFValue): void {
    if (this.ui.change) {
      this.ui.change(values, this.getOrgData(values));
    }
    this.setValue(values == null ? undefined : values);
  }

  private getOrgData(values: DFValue): DFSchemaEnum | DFSchemaEnum[] {
    const srv = this.injector.get(ArrayService);
    if (!Array.isArray(values)) {
      return srv.findTree(
        this.data,
        (item: DFSchemaEnum) => item.value === values
      )!;
    }
    return values.map((value) =>
      srv.findTree(this.data, (item: DFSchemaEnum) => item.value === value)
    );
  }

  openChange(status: boolean): void {
    if (this.ui.openChange) {
      this.ui.openChange(status);
    }
  }

  scrollToBottom(): void {
    if (this.ui.scrollToBottom) {
      this.ui.scrollToBottom();
    }
  }

  onSearch(value: string): void {
    if (this.ui.onSearch) {
      this.loading = true;
      this.search$.next(value);
    }
  }
}
```

```ts
export function getData(
  schema: DFSchema,
  ui: DFUISchemaItem,
  formData: NzSafeAny,
  asyncArgs?: NzSafeAny
): Observable<DFSchemaEnum[]> {
  if (typeof ui.asyncData === "function") {
    // 异步数据源
    return ui
      .asyncData(asyncArgs)
      .pipe(
        map((list: DFSchemaEnum[]) => getEnum(list, formData, schema.readOnly!))
      );
  }
  // 静态数据源
  return of(getCopyEnum(schema.enum!, formData, schema.readOnly!));
```

```ts
@ViewChild('df', { static: true }) df!: TlDfComponent;
    private searchDepartment$ = new Subject<string>();
    tlSchema: DFSchema = {
        properties: {

                type: 'string',
                title: '登录方式',
                enum: [
                    { label: '手机', value: 'mobile' },
                    { label: '账密', value: 'account' }

                    widget: 'radio',
                    styleType: 'button'
                }
            },
            mobile: {
                type: 'string',
                description: '请尝试输入: topdsm',

                    changeDebounceTime: 100,
                    change: q => {
                        this.df.getProperty('/department')?.updateFeedback('validating');
                        this.searchDepartment$.next(q);
                    },
                    visibleIf: {
                        login_type: val => (val === 'mobile' ? { required: true, show: true } : null)
                    }
                } as DFStringWidgetSchema
            },
            code: {
                type: 'number',
                ui: {
                    visibleIf: {
                        login_type: ['mobile']

                }
            },

                }

                    visibleIf: {
```
